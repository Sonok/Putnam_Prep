\begin{problem}
Find all positive integers $n$ such that $n$ divides $2^n - 1$.
\end{problem}

\vspace{10pt}\hrule

\clearpage
\section*{Hint}
Not before 25 minutes.

\spoilerspace
\begin{hint}
Suppose $n > 1$ works. Look at the \emph{smallest} prime $p$ dividing $n$. You know $2^n \equiv 1 \pmod p$. Fermat's little theorem gives a second exponent that also works.
\end{hint}
